// ఫైల్‌ను నేరుగా తాత్కాలిక పబ్లిక్ లింక్ గా మార్చడం (GitHub commits అవసరం లేదు) async function uploadDocumentToGitHub(file) { try { const formData = new FormData(); formData.append('reqtype', 'fileupload'); formData.append('fileToUpload', file); const res = await fetch('https://catbox.moe/user/api.php', { method: 'POST', body: formData }); if (res.ok) { const directUrl = await res.text(); return directUrl.trim(); } } catch (e) { console.error("Direct upload error:", e); } return ""; }
